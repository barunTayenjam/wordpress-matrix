  ### B. Add --json flag parsing to health-check.sh

  At the top of `scripts/health-check.sh`, right after the shebang and `set -euo pipefail` (line 4), add:

  ```bash
  # Parse arguments
  JSON_OUTPUT=false
  while [[ $# -gt 0 ]]; do
      case "$1" in
          --json)
              JSON_OUTPUT=true
              shift
              ;;
          *)
              echo "Usage: $0 [--json]" >&2
              exit 1
              ;;
      esac
  done
  ```

  Then, at the very end of the script (after all the human-readable output), add the JSON output block. When `JSON_OUTPUT` is true, the script should:

  1. Collect all the same data the human-readable path collects
  2. Emit a single JSON object to stdout
  3. **NOT** emit any human-readable output (no echo/log_info/log_success/log_error calls)

  Add this block at the END of `scripts/health-check.sh` (after line 161, before any trailing newlines):

  ```bash
  # JSON output mode
  if [[ "$JSON_OUTPUT" == "true" ]]; then
      # Collect site statuses
      local json_sites="["
      local first=true
      for site in $(get_sites); do
          if [[ "$first" == "true" ]]; then
              first=false
          else
              json_sites+=","
          fi
          local running
          if $CONTAINER_RUNTIME ps --format "{{.Names}}" 2>/dev/null | grep -q "^wp_$site$"; then
              json_sites+="{\"name\":\"$site\",\"running\":true}"
          else
              json_sites+="{\"name\":\"$site\",\"running\":false}"
          fi
      done
      json_sites+="]"

      # Determine db and redis status
      local db_running=false
      local db_conn=false
      local redis_running=false
      local redis_conn=false

      if $CONTAINER_RUNTIME ps --format "{{.Names}}" 2>/dev/null | grep -q "^wp_db$"; then
          db_running=true
          if $CONTAINER_RUNTIME exec wp_db mysqladmin ping -h localhost --silent 2>/dev/null; then
              db_conn=true
          fi
      fi
      if $CONTAINER_RUNTIME ps --format "{{.Names}}" 2>/dev/null | grep -q "^wp_redis$"; then
          redis_running=true
          if $CONTAINER_RUNTIME exec wp_redis redis-cli ping 2>/dev/null | grep -q "PONG"; then
              redis_conn=true
          fi
      fi

      # Determine runtime
      local runtime_name
      if command -v podman >/dev/null 2>&1; then
          runtime_name="podman"
      else
          runtime_name="docker"
      fi

      # Determine compose tool
      local compose_tool
      if command -v podman-compose >/dev/null 2>&1; then
          compose_tool="podman-compose"
      elif docker compose version >/dev/null 2>&1; then
          compose_tool="docker-compose"
      else
          compose_tool="docker-compose (standalone)"
      fi

      # Count running containers
      local containers_running
      containers_running=$($CONTAINER_RUNTIME ps --format "{{.Names}}" 2>/dev/null | wc -l | tr -d ' ')

      # Disk info
      local df_output disk_usage_percent disk_available
      df_output=$(df -h "$PROJECT_ROOT" 2>/dev/null | tail -1)
      disk_usage_percent=$(echo "$df_output" | awk '{print $5}' 2>/dev/null || echo "0%")
      disk_available=$(echo "$df_output" | awk '{print $4}' 2>/dev/null || echo "unknown")

      cat << JSONEOF
  {
    "success": true,
    "data": {
      "containerRuntime": "${runtime_name}",
      "composeTool": "${compose_tool}",
      "containersRunning": ${containers_running},
      "database": {
        "containerRunning": ${db_running},
        "connectionOk": ${db_conn}
      },
      "redis": {
        "containerRunning": ${redis_running},
        "connectionOk": ${redis_conn}
      },
      "sites": ${json_sites},
      "diskUsagePercent": "${disk_usage_percent}",
      "diskAvailable": "${disk_available}",
      "errorCount": ${error_count}
    }
  }
  JSONEOF
      exit 0
  fi
  ```

  Note: The `get_sites()` function is already available since health-check.sh sources common.sh (which sources helpers.sh where get_sites is defined).

  The JSON structure should match existing patterns from `matrix status --json` and `matrix info --json`. Check matrix's status output for the exact key naming convention and match it.
